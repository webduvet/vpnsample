# /etc/ipsec.d/vgw.conf - libreSwan configuration file for AWS VGW

conn %default
  ike=aes256-sha256-dh19
  esp=aes256-sha256-dh19
  pfs=yes
  ikelifetime=14400s
  salifetime=3600s
  dpddelay=30s
  dpdtimeout=120s

  # VTI
  mark=42/0xffffffff
  vti-interface=vti0
  vti-shared=yes
  vti-routing=no    # we add explicit routes below

  # Wide selectors keep VGW happy; we control real traffic via routes
  left=%defaultroute
  leftid=18.202.63.87
  leftsubnet=0.0.0.0/0
  rightsubnet=0.0.0.0/0
  type=tunnel

conn vgw1
  right=52.211.164.164
  auto=start
  authby=secret

conn vgw2
  also=vgw1
  right=54.217.2.73
  auto=start
  authby=secret



  # nft rules for VTI interface
[ec2-user@ip-10-200-0-253 ~]$ sudo nft list ruleset
table ip nat {
        chain PREROUTING {
                type nat hook prerouting priority dstnat; policy accept;
        }

        chain POSTROUTING {
                type nat hook postrouting priority srcnat; policy accept;
        }

        chain prerouting {
                type nat hook prerouting priority dstnat; policy accept;
                iif "vti0" ip daddr 203.0.113.246 tcp dport 8080 dnat to 10.200.1.246
        }

        chain postrouting {
                type nat hook postrouting priority srcnat; policy accept;
                oif "vti0" ip saddr 10.200.1.246 tcp sport 8080 snat to 203.0.113.246
        }
}
table ip filter {
        chain FORWARD {
                type filter hook forward priority filter; policy accept;
                iifname "ens5" oifname "vti0" counter packets 21 bytes 3801 accept
                iifname "vti0" oifname "ens5" counter packets 24 bytes 1533 accept
                ct state related,established counter packets 0 bytes 0 accept
        }
}
